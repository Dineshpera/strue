Express 5 server with a JSON route and a wildcard route
express5-hello

Create an Express server with a JSON endpoint and a catch-all path segment. Official docs · recipe file

Steps, run from an empty folder

npm init -y
npm install express

app.mjs

import express from "express";

const app = express();
app.get("/", (req, res) => res.json({ message: "Hello World" }));
app.get("/files/*splat", (req, res) => res.json({ path: req.params.splat }));

const server = app.listen(0, async () => {
  const { port } = server.address();
  const root = await (await fetch(`http://127.0.0.1:${port}/`)).json();
  const file = await (await fetch(`http://127.0.0.1:${port}/files/a/b.txt`)).json();
  server.close();
  if (root.message !== "Hello World") throw new Error("root route failed");
  if (file.path.join("/") !== "a/b.txt") throw new Error("wildcard route failed");
  console.log("strue-ok");
});

  • node stale.mjs — Express 5 rejects a bare "*" path; wildcards need a name, such as "/*splat". confirmed broken

MCP server in Python with one tool, called by an in-process client
mcp-python-server

Write an MCP server with the official Python SDK and call its tool. Official docs · recipe file

Steps, run from an empty folder

pip install mcp

check.py

import anyio
from mcp import Client
from mcp.server.mcpserver import MCPServer

server = MCPServer("demo")


@server.tool()
def add(a: int, b: int) -> int:
    """Add two numbers."""
    return a + b


async def main() -> None:
    async with Client(server) as client:
        tools = await client.list_tools()
        assert [tool.name for tool in tools.tools] == ["add"]
        result = await client.call_tool("add", {"a": 2, "b": 3})
        assert result.content[0].text == "5", result
    print("strue-ok")


anyio.run(main)

pip install pydantic

check.py

from pydantic import BaseModel, ValidationError, field_validator


class User(BaseModel):
    id: int
    name: str

    @field_validator("name")
    @classmethod
    def name_not_blank(cls, value: str) -> str:
        if not value.strip():
            raise ValueError("name must not be blank")
        return value


user = User.model_validate({"id": "7", "name": "Asha"})
assert user.id == 7
assert user.model_dump() == {"id": 7, "name": "Asha"}
assert User.model_validate_json(user.model_dump_json()) == user
try:
    User.model_validate({"id": 1, "name": " "})
except ValidationError:
    print("strue-ok")
